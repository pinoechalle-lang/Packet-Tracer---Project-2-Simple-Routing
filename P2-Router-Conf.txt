=============================================================================
R1
=============================================================================

[ Escalate Mode ]

enable
configure terminal


[ Host Name and Banner ]

hostname R1
banner motd #
--- ROUTER 1: AUTHORIZED PERSONNEL ONLY! ---
#

no ip domain-lookup


[ Security ]

line console 0
 password hi
 login
 exec-timeout 5
exit

enable secret hello

service password-encryption

login block-for 60 attempts 3 within 60


[ Facing VLAN 1 (LAN A) ]

interface g0/0
 description Link to LAN A
 ip address 192.168.50.254 255.255.255.0
 no shutdown
exit


[ Facing Router 2 ]

interface g0/1
 description Link to R2
 ip address 172.27.39.1 255.255.255.252
 no shutdown
exit


[ IP Routing ]

ip routing
ip route 192.168.66.0 255.255.255.0 172.27.39.2


[ Save (Global) ]

do copy running-config startup-config
